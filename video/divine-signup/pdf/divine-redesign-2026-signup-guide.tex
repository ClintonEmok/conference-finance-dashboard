% Step pages
\stepPage{1}{Start with the event signup}{Begin with the Divine Redesign 2026 registration flow. This example follows one attendee selecting a Single Room ticket. The same steps apply to families and other configurations; choose the options that match your group.}{../out/pdf-stills/01-welcome.png}
\infoBox{%
  \eyebrow{What this step covers}\par\vspace{0.08in}
  The walkthrough follows the live flow from ticket selection through accommodation choices, review, verification, and payment guidance.
}

\stepPage{2}{Select one ticket}{Choose the ticket category and quantity for the attendee. In this example, select one Single Room ticket.}{../out/pdf-stills/02-tickets.png}
\infoBox{%
  \eyebrow{Example selection}\par\vspace{0.08in}
  One Single Room ticket at \texteuro\,350.00. Use the quantity controls if you are registering more than one attendee.
}

\stepPage{3}{Enter buyer and attendee details}{Complete the buyer contact details, then enter the attendee information required for the registration.}{../out/pdf-stills/03-details.png}
\infoBox{%
  \eyebrow{Have these ready}\par\vspace{0.08in}
  Full name, email address, phone number, gender, location, dietary information, and any relevant room preferences.
}

\stepPage{4}{Review the included stay}{The Single Room ticket includes a Standard single room for two nights. The final room placement is confirmed by the organizer.}{../out/pdf-stills/04-included-stay.png}
\infoBox{%
  \eyebrow{Included accommodation}\par\vspace{0.08in}
  The included stay covers two nights. Breakfast is associated with the separately selected night-before stay, not the included accommodation.
}

\stepPage{5}{Add accommodation options}{On the accommodation step, add a cot by choosing its quantity and number of nights. You can also upgrade the included stay to Superior.}{../out/pdf-stills/05-options.png}
\infoBox{%
  \eyebrow{Example options}\par\vspace{0.08in}
  One cot for two nights: \texteuro\,20.00. Superior upgrade for one attendee across two nights: \texteuro\,20.00.
}

\stepPage{6}{Choose a night-before stay}{If you want to come a night before, choose Standard or Superior, then Single or Shared occupancy.}{../out/pdf-stills/06-night-before.png}
\infoBox{%
  \eyebrow{Example selection}\par\vspace{0.08in}
  The example uses Superior \textperiodcentered\ Shared at \texteuro\,70.00. Breakfast is included with the selected night-before stay.
}

\stepPage{7}{Review before submitting}{Check the buyer, attendee, ticket, and every accommodation charge before completing the verification challenge.}{../out/pdf-stills/07-review.png}
\infoBox{%
  \eyebrow{Final review checklist}\par\vspace{0.08in}
  Confirm the live total, attendee count, accommodation selections, dates, and contact details before submitting.
}

\stepPage{8}{Submit and manage the booking}{After submission, save the booking reference and complete payment through Tikkie. A confirmation email will arrive.}{../out/pdf-stills/08-confirmation.png}
\infoBox{%
  \eyebrow{After submission}\par\vspace{0.08in}
  Use Manage Booking later to review payment progress and booking details. The walkthrough itself does not create a production booking.
}
