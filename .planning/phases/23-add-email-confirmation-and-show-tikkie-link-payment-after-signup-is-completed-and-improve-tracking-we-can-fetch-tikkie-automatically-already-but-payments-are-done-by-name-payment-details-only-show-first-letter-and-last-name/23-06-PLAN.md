```typescript
async function handleSubmitFromReview() {
  // ... existing validation ...

  setIsSubmitting(true)
  setSubmitError(null)

  const result = await submitSignupDraft(activeDraft, { idempotencyKey })

  if (!result.ok) {
    setSubmitResult(null)
    setRestoreChoicePending(false)
    setSubmitError(result.error)
    setIsSubmitting(false)
    return
  }

  // Clear localStorage draft on successful submission
  if (eventId) {
    try {
      window.localStorage.removeItem(`signup-draft:${eventId}`)
    } catch {
      // Ignore localStorage errors
    }
  }

  setSubmitResult(result.data)
  setSubmitError(null)
  setRestoreChoicePending(Boolean(result.data.restorePayload))
  setIsSubmitting(false)
}
```

The key change is adding the localStorage.removeItem call right after successful submission, before updating the submitResult state.
